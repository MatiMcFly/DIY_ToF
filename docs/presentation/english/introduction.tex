\begin{frame}{ToF}
    \begin{itemize}
        \item ToF = Time of Flight
        \item It is the time it takes for a signal to travel from the sender to the receiver
    \end{itemize}
\end{frame}

\begin{frame}{ToF: Distance measurement}
    \begin{columns}
